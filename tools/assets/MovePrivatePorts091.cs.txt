using System;using System.IO;using System.Linq;using System.Text;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class MovePrivatePorts091 {
 static bool OnGrass(Transform island,Vector3 world){var mf=island.Find("Grass_Terrace").GetComponent<MeshFilter>();var p=mf.transform.InverseTransformPoint(world);var vv=mf.sharedMesh.vertices;var tr=mf.sharedMesh.triangles;for(int i=0;i<tr.Length;i+=3){var a=vv[tr[i]];var b=vv[tr[i+1]];var c=vv[tr[i+2]];float den=(b.z-c.z)*(a.x-c.x)+(c.x-b.x)*(a.z-c.z);float u=((b.z-c.z)*(p.x-c.x)+(c.x-b.x)*(p.z-c.z))/den;float v=((c.z-a.z)*(p.x-c.x)+(a.x-c.x)*(p.z-c.z))/den;if(u>=-.001f&&v>=-.001f&&u+v<=1.001f)return true;}return false;}
 static List<Vector3> Vertices(Transform boat){var list=new List<Vector3>();foreach(var mf in boat.GetComponentsInChildren<MeshFilter>().Where(m=>m.gameObject.activeInHierarchy&&m.sharedMesh!=null))list.AddRange(mf.sharedMesh.vertices.Select(v=>mf.transform.TransformPoint(v)));return list;}
 static void Shot(Vector3 pos,Vector3 target,float size,string name){var go=new GameObject("Temporary_Port091_Camera");var rt=new RenderTexture(1800,1200,24);var tex=new Texture2D(1800,1200,TextureFormat.RGB24,false);var old=RenderTexture.active;try{var c=go.AddComponent<Camera>();c.transform.position=pos;c.transform.LookAt(target);c.orthographic=true;c.orthographicSize=size;c.farClipPlane=600;c.clearFlags=CameraClearFlags.SolidColor;c.backgroundColor=new Color(.19f,.39f,.43f);c.targetTexture=rt;c.Render();RenderTexture.active=rt;tex.ReadPixels(new Rect(0,0,1800,1200),0,0);tex.Apply();File.WriteAllBytes("Docs/assets/lobby-island/"+name+".png",tex.EncodeToPNG());}finally{RenderTexture.active=old;UnityEngine.Object.DestroyImmediate(go);UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(tex);}}
 public static string Run(){var s=UnityEngine.SceneManagement.SceneManager.GetActiveScene();if(s.path!="Assets/Scenes/LobbyIslandWorkshop.unity"||s.isDirty||EditorApplication.isPlayingOrWillChangePlaymode)throw new Exception("Saved prototype EditMode required");var root=GameObject.Find("Lobby_Island_Visual_Prototype").transform;var main=root.Find("01_Central_Island");var plots=root.Find("02_Personal_Islands_8_Visual_Plots").Cast<Transform>().ToArray();if(plots.Length!=8||plots.Any(p=>p.Find("04_Private_Dock").localRotation!=Quaternion.identity))throw new Exception("Expected original eight port positions");
 // Check candidate placement on copies of mesh coordinates before mutating the scene.
 foreach(var p in plots){var d=p.Find("04_Private_Dock");var boat=d.Find("Actual_BoardController_RenderOnly_1to1");foreach(var w in Vertices(boat)){var q=d.InverseTransformPoint(w);q=new Vector3(-q.x,q.y,-q.z-5);var candidate=p.TransformPoint(q);if(plots.Any(t=>OnGrass(t,candidate))||OnGrass(main,candidate))throw new Exception("Candidate boat intersects shore: "+p.name+" "+q);if(q.x> -2.6f)throw new Exception("Boat too close to bridge");}}
 var report=new StringBuilder();var allBoatPoints=new List<List<Vector3>>();for(int i=0;i<plots.Length;i++){var p=plots[i];var d=p.Find("04_Private_Dock");d.localPosition=new Vector3(0,0,-5);d.localRotation=Quaternion.Euler(0,180,0);d.Find("Compact_Private_Dock089").gameObject.SetActive(false);var branch=d.Find("Compact_Private_Dock_Return089");branch.name="Inner_Private_Port_Branch091";foreach(var post in branch.Cast<Transform>().Where(t=>t.name.Contains("Post")||t.name.Contains("Cap")).ToArray())if(p.InverseTransformPoint(post.position).x> -1.5f)post.gameObject.SetActive(false);
 var bridge=root.Find("08_Ring_Connections090/Personal_Bridge_"+(i+1));foreach(var post in bridge.Cast<Transform>().Where(t=>t.name.Contains("Post")||t.name.Contains("Cap")).ToArray()){var q=p.InverseTransformPoint(post.position);if(q.x<0&&Mathf.Abs(q.z-37)<1.8f)post.gameObject.SetActive(false);}d.Find("Future_Dock_Interaction_Anchor").localPosition=new Vector3(0,0,-29);var pen=p.Find("08_Large_Animal_Pen090");foreach(var t in pen.Cast<Transform>().Where(t=>t.name.StartsWith("Fence_")&&t.localPosition.z<-14))t.gameObject.SetActive(true);pen.Find("Future_Private_Dock_Gate_Anchor").gameObject.SetActive(false);
 var path=p.Cast<Transform>().Single(t=>t.name=="New_Path090"&&t.localPosition.z<0&&t.gameObject.activeSelf);path.localPosition=new Vector3(0,.05f,-2);path.localScale=new Vector3(2.2f,.07f,20);var points=Vertices(d.Find("Actual_BoardController_RenderOnly_1to1"));allBoatPoints.Add(points);var local=points.Select(p.InverseTransformPoint).ToArray();float minX=local.Min(v=>v.x),maxX=local.Max(v=>v.x),minZ=local.Min(v=>v.z),maxZ=local.Max(v=>v.z);if(maxX>-2.6f||minZ<18||maxZ>37)throw new Exception("Berth fit failed");report.AppendLine(p.name+": inner berth boat x["+minX.ToString("F2")+","+maxX.ToString("F2")+"] z["+minZ.ToString("F2")+","+maxZ.ToString("F2")+"]; bridge edge gap "+(-maxX-1.2f).ToString("F2")+"m; branch at z37 x[-11,0], 2.6m wide; original boat scale retained; all active hull vertices outside grass triangles.");}
 var exhibit=root.Find("04_Boat_Skins_Showcase/Actual_Boat_RenderOnly_1to1");allBoatPoints.Add(Vertices(exhibit));float minGap=float.MaxValue;for(int i=0;i<allBoatPoints.Count;i++)for(int j=i+1;j<allBoatPoints.Count;j++){var a=allBoatPoints[i];var b=allBoatPoints[j];float dx=Mathf.Max(0,Mathf.Max(a.Min(v=>v.x)-b.Max(v=>v.x),b.Min(v=>v.x)-a.Max(v=>v.x)));float dz=Mathf.Max(0,Mathf.Max(a.Min(v=>v.z)-b.Max(v=>v.z),b.Min(v=>v.z)-a.Max(v=>v.z)));float gap=Mathf.Sqrt(dx*dx+dz*dz);minGap=Mathf.Min(minGap,gap);if(gap<1)throw new Exception("Boat bounds overlap");}if(root.GetComponentsInChildren<Collider>(true).Length>0||root.GetComponentsInChildren<Rigidbody>(true).Length>0)throw new Exception("Unexpected physics");report.AppendLine("All 8 personal ports moved inward; shared bridge used for access; minimum separation of world axis-aligned bounds among nine boats = "+minGap.ToString("F2")+"m. Old outward longitudinal dock hidden; rear fence restored and obsolete rear approach shortened. No gameplay logic added.");Shot(new Vector3(150,230,-210),main.position,115,"10-inner-ports-overview091");Shot(plots[4].TransformPoint(new Vector3(45,60,48)),plots[4].TransformPoint(new Vector3(-3,0,26)),32,"11-inner-private-port091");EditorSceneManager.MarkSceneDirty(s);if(!EditorSceneManager.SaveScene(s))throw new Exception("Save failed");File.WriteAllText("Docs/assets/lobby-island/PORTS091_VALIDATION.txt",report.ToString());var cam=root.GetComponentInChildren<Camera>(true);cam.orthographicSize=115;EditorSceneManager.MarkSceneDirty(s);EditorSceneManager.SaveScene(s);return report.ToString()+"Scene saved clean="+!s.isDirty;}
}
