 static Vector2 Slice111(List<Vector2> hull,float z){var xs=new List<float>();for(int i=0;i<hull.Count;i++){var a=hull[i];var b=hull[(i+1)%hull.Count];if(z<Mathf.Min(a.y,b.y)-.0001f||z>Mathf.Max(a.y,b.y)+.0001f)continue;if(Mathf.Abs(a.y-b.y)<.00001f){xs.Add(a.x);xs.Add(b.x);}else xs.Add(Mathf.Lerp(a.x,b.x,(z-a.y)/(b.y-a.y)));}if(xs.Count==0)throw new Exception("Fuel z outside hull "+z);return new Vector2(xs.Min(),xs.Max());}
