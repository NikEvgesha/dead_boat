using System;using System.IO;using System.Linq;using System.Text;using System.Collections.Generic;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class FixRocksShore093{
 const string D="Assets/Prefabs/Workshop/LobbyIsland/";static Material oak,dark,gold,gravel,rock;static Dictionary<Transform,List<Vector2>> shores=new Dictionary<Transform,List<Vector2>>();static StringBuilder report=new StringBuilder();
 static Transform G(string n,Transform p,Vector3 v){var t=new GameObject(n).transform;t.SetParent(p,false);t.localPosition=v;return t;}
 static GameObject Box(string n,Transform p,Vector3 v,Vector3 size,Material mat){var g=GameObject.CreatePrimitive(PrimitiveType.Cube);g.name=n;g.transform.SetParent(p,false);g.transform.localPosition=v;g.transform.localScale=size;g.GetComponent<Renderer>().sharedMaterial=mat;UnityEngine.Object.DestroyImmediate(g.GetComponent<Collider>());return g;}
 static List<Vector2> Shore(Transform p){var mf=p.Find("Soft_Sand_Beach").GetComponent<MeshFilter>();return mf.sharedMesh.vertices.Where((v,i)=>i%2==1).Select(v=>mf.transform.TransformPoint(v)).Select(v=>new Vector2(v.x,v.z)).ToList();}
 static bool Inside(List<Vector2> poly,Vector2 q){bool inside=false;for(int i=0,j=poly.Count-1;i<poly.Count;j=i++){var a=poly[i];var b=poly[j];if((a.y>q.y)!=(b.y>q.y)&&q.x<(b.x-a.x)*(q.y-a.y)/(b.y-a.y)+a.x)inside=!inside;}return inside;}
 static bool Land(Vector3 w){var q=new Vector2(w.x,w.z);return shores.Values.Any(p=>Inside(p,q));}
 static float PointSegment(Vector2 q,Vector2 a,Vector2 b){var d=b-a;float t=Mathf.Clamp01(Vector2.Dot(q-a,d)/d.sqrMagnitude);return Vector2.Distance(q,a+d*t);}
 static float Clearance(List<Vector2> a,List<Vector2> b){if(a.Any(q=>Inside(b,q))||b.Any(q=>Inside(a,q)))return 0;float d=float.MaxValue;foreach(var q in a)for(int i=0;i<b.Count;i++)d=Mathf.Min(d,PointSegment(q,b[i],b[(i+1)%b.Count]));foreach(var q in b)for(int i=0;i<a.Count;i++)d=Mathf.Min(d,PointSegment(q,a[i],a[(i+1)%a.Count]));return d;}
 static List<Vector3> V(Transform t){return t.GetComponentsInChildren<MeshFilter>().Where(m=>m.gameObject.activeInHierarchy&&m.sharedMesh!=null).SelectMany(m=>m.sharedMesh.vertices.Select(v=>m.transform.TransformPoint(v))).ToList();}
 static bool ClearRect(Transform p,float x0,float x1,float z0,float z1){for(float x=x0;x<=x1+.001f;x+=.4f)foreach(float z in new[]{z0,z1})if(Land(p.TransformPoint(new Vector3(x,0,z))))return false;for(float z=z0;z<=z1+.001f;z+=.4f)foreach(float x in new[]{x0,x1})if(Land(p.TransformPoint(new Vector3(x,0,z))))return false;return true;}
 static Transform Deck(string n,Transform p,Vector3 a,Vector3 b,float width,float branchZ=float.NaN){var v=b-a;var t=G(n,p,(a+b)/2);t.localRotation=Quaternion.LookRotation(v);int count=Mathf.CeilToInt(v.magnitude/.5f);for(int i=0;i<count;i++)Box("Deck_Plank",t,new Vector3(0,-.1f,-v.magnitude/2+(i+.5f)*v.magnitude/count),new Vector3(width,.2f,v.magnitude/count-.03f),oak);for(float z=-v.magnitude/2;z<v.magnitude/2;z+=3)foreach(float x in new[]{-width/2+.2f,width/2-.2f}){var point=t.TransformPoint(new Vector3(x,0,z));if(!float.IsNaN(branchZ)&&x<0&&Mathf.Abs(p.InverseTransformPoint(point).z-branchZ)<1.1f)continue;Box("Pier_Post",t,new Vector3(x,-1.05f,z),new Vector3(.2f,3,.2f),dark);Box("Brass_Cap",t,new Vector3(x,.5f,z),new Vector3(.24f,.07f,.24f),gold);}return t;}
 static void Path(Transform p,Vector3 a,Vector3 b){var t=Box("Arrival_Path092",p,(a+b)/2+Vector3.up*.05f,new Vector3(2.4f,.07f,(b-a).magnitude),gravel).transform;t.localRotation=Quaternion.LookRotation(b-a);}
 static bool Grass(Transform p,Vector3 q){var mf=p.Find("Grass_Terrace").GetComponent<MeshFilter>();var outline=mf.sharedMesh.vertices.Skip(1).Select(v=>mf.transform.TransformPoint(v)).Select(v=>new Vector2(v.x,v.z)).ToList();return Inside(outline,new Vector2(q.x,q.z));}
 static Transform Copy(Transform src,Transform p){var t=G(src.name,p,src.localPosition);t.localRotation=src.localRotation;t.localScale=src.localScale;var mf=src.GetComponent<MeshFilter>();var mr=src.GetComponent<MeshRenderer>();if(mf!=null&&mr!=null){t.gameObject.AddComponent<MeshFilter>().sharedMesh=mf.sharedMesh;t.gameObject.AddComponent<MeshRenderer>().sharedMaterials=mr.sharedMaterials;}foreach(Transform c in src)Copy(c,t);t.gameObject.SetActive(src.gameObject.activeSelf);return t;}
 static Transform Visual(string asset,string n,Transform p,Vector3 pos,float size,float yaw){var a=AssetDatabase.LoadAssetAtPath<GameObject>(asset);var t=Copy(a.transform,p);t.name=n;t.gameObject.SetActive(true);t.localPosition=Vector3.zero;t.localRotation=Quaternion.Euler(0,yaw,0);var vv=V(t).Select(p.InverseTransformPoint).ToArray();var min=new Vector3(vv.Min(v=>v.x),vv.Min(v=>v.y),vv.Min(v=>v.z));var max=new Vector3(vv.Max(v=>v.x),vv.Max(v=>v.y),vv.Max(v=>v.z));t.localScale*=size/Mathf.Max((max-min).x,Mathf.Max((max-min).y,(max-min).z));vv=V(t).Select(p.InverseTransformPoint).ToArray();t.localPosition+=pos-new Vector3((vv.Min(v=>v.x)+vv.Max(v=>v.x))/2,vv.Min(v=>v.y),(vv.Min(v=>v.z)+vv.Max(v=>v.z))/2);return t;}
 static void Palm(Transform decor,Vector3 pos,float size,float yaw){Visual("Assets/Prefabs/Locations/Decoration/Beach_Prop_Tree_Palm_2.prefab","Palm_UniformScale092",decor,pos,size,yaw);}
 static void Bush(Transform decor,Vector3 pos,float size,float yaw){Visual("Assets/Prefabs/Locations/Decoration/Hilly_Prop_Bush_1.prefab","Shrub092",decor,pos,size,yaw);}
 static void Rocks(Transform d,Vector3 v,int seed){for(int i=0;i<3;i++){var g=GameObject.CreatePrimitive(PrimitiveType.Sphere);g.name="Pebble_Cluster092";g.transform.SetParent(d,false);g.transform.localPosition=v+new Vector3(i*.55f,.18f,-i*.22f);g.transform.localScale=new Vector3(.9f+i*.15f,.45f+i*.08f,.65f);g.transform.localRotation=Quaternion.Euler(7,seed+i*37,13);g.GetComponent<Renderer>().sharedMaterial=rock;UnityEngine.Object.DestroyImmediate(g.GetComponent<Collider>());}}
 static void Shot(Vector3 pos,Vector3 target,float size,string name){var go=new GameObject("Temporary_092_Camera");var rt=new RenderTexture(1800,1200,24);var tex=new Texture2D(1800,1200,TextureFormat.RGB24,false);var old=RenderTexture.active;try{var c=go.AddComponent<Camera>();c.transform.position=pos;c.transform.LookAt(target);c.orthographic=true;c.orthographicSize=size;c.farClipPlane=600;c.clearFlags=CameraClearFlags.SolidColor;c.backgroundColor=new Color(.19f,.39f,.43f);c.targetTexture=rt;c.Render();RenderTexture.active=rt;tex.ReadPixels(new Rect(0,0,1800,1200),0,0);tex.Apply();File.WriteAllBytes("Docs/assets/lobby-island/"+name+".png",tex.EncodeToPNG());}finally{RenderTexture.active=old;UnityEngine.Object.DestroyImmediate(go);UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(tex);}}
 static List<Vector2> Foam(Transform p){var mf=p.Find("Foam_Shoreline").GetComponent<MeshFilter>();return mf.sharedMesh.vertices.Where((v,i)=>i%2==1).Select(v=>mf.transform.TransformPoint(v)).Select(v=>new Vector2(v.x,v.z)).ToList();}
 static List<Vector2> HullBox(Transform boat,Transform plot){var points=V(boat).Select(plot.InverseTransformPoint).ToArray();float x0=points.Min(v=>v.x),x1=points.Max(v=>v.x),z0=points.Min(v=>v.z),z1=points.Max(v=>v.z);return new[]{new Vector3(x0,0,z0),new Vector3(x1,0,z0),new Vector3(x1,0,z1),new Vector3(x0,0,z1)}.Select(plot.TransformPoint).Select(v=>new Vector2(v.x,v.z)).ToList();}
 public static string Run(){var s=UnityEngine.SceneManagement.SceneManager.GetActiveScene();if(s.path!="Assets/Scenes/LobbyIslandWorkshop.unity"||s.isDirty||EditorApplication.isPlayingOrWillChangePlaymode)throw new Exception("Saved prototype EditMode required");var root=GameObject.Find("Lobby_Island_Visual_Prototype").transform;var main=root.Find("01_Central_Island");var plots=root.Find("02_Personal_Islands_8_Visual_Plots").Cast<Transform>().ToArray();if(root.GetComponentsInChildren<Transform>(true).Any(t=>t.name=="Angular_Rock093"))throw new Exception("Already applied");var template=root.GetComponentsInChildren<Transform>(true).First(t=>t.name=="Faceted_Shore_Rock");var rounds=root.GetComponentsInChildren<Transform>(true).Where(t=>t.name=="Pebble_Cluster092").ToArray();if(rounds.Length!=78)throw new Exception("Unexpected round stones count "+rounds.Length);foreach(var round in rounds){var p=round.parent;var v=round.localPosition;var copy=Copy(template,p);copy.name="Angular_Rock093";copy.localPosition=new Vector3(v.x,.25f,v.z);copy.localRotation=Quaternion.Euler(10,round.localEulerAngles.y,15);copy.localScale=template.localScale*(.72f+((int)Mathf.Abs(v.x*3+v.z)%3)*.10f);UnityEngine.Object.DestroyImmediate(round.gameObject);}report.AppendLine("Removed all "+rounds.Length+" round Pebble_Cluster092; replaced with static copies of existing Faceted_Shore_Rock mesh/material, 48 across personal plots +30 on main island; original square shore stones retained.");oak=AssetDatabase.LoadAssetAtPath<Material>(D+"Oiled_Oak.mat");dark=AssetDatabase.LoadAssetAtPath<Material>(D+"Deep_Charcoal.mat");gold=AssetDatabase.LoadAssetAtPath<Material>(D+"Brass_Gold.mat");var coast=Foam(main);
 for(int i=0;i<plots.Length;i++){var p=plots[i];var boat=p.Find("04_Private_Dock/Actual_BoardController_RenderOnly_1to1");var bridge=p.Find("Plot_"+(i+1)+"_Bridges092/Main_Island_Bridge092");var oldEnd=bridge.TransformPoint(new Vector3(0,0,bridge.Cast<Transform>().Where(t=>t.name=="Deck_Plank").Max(t=>t.localPosition.z+t.localScale.z/2)));var outward=(p.position-main.position).normalized;float before=Clearance(HullBox(boat,p),coast);float shift=0;while(Clearance(HullBox(boat,p),coast)<2.1f&&shift<9){p.position+=outward*.5f;shift+=.5f;}float after=Clearance(HullBox(boat,p),coast);if(after<2.1f)throw new Exception("Cannot create safe shore gap");if(shift>0){bridge.gameObject.SetActive(false);var plink=bridge.parent;var branch=p.Find("Plot_"+(i+1)+"_Bridges092/Private_Port_Branch092");float bz=p.InverseTransformPoint(branch.position).z;var updated=Deck("Main_Island_Bridge093",plink,new Vector3(0,0,16),p.InverseTransformPoint(oldEnd),2.4f,bz);foreach(var plank in updated.Cast<Transform>().Where(t=>t.name=="Deck_Plank")){var v=plank.localPosition;v.y=-.03f;plank.localPosition=v;}}report.AppendLine(p.name+": complete hull bounding footprint to outer foam shore "+before.ToString("F2")+" -> "+after.ToString("F2")+"m; entire plot/boat/pier moved outward "+shift.ToString("F1")+"m, main shore bridge endpoint kept.");}
 shores.Clear();shores[main]=Foam(main);foreach(var p in plots)shores[p]=Foam(p);float minIslands=float.MaxValue;for(int i=0;i<plots.Length;i++)for(int j=i+1;j<plots.Length;j++){float gap=Clearance(shores[plots[i]],shores[plots[j]]);if(gap<8)throw new Exception("Personal shore gap too small");minIslands=Mathf.Min(minIslands,gap);}foreach(var p in plots){var boat=p.Find("04_Private_Dock/Actual_BoardController_RenderOnly_1to1");if(boat.localScale!=Vector3.one||boat.Find("TrailerDefault").gameObject.activeInHierarchy)throw new Exception("Boat scale/trailer changed");foreach(var v in V(boat))if(Land(v))throw new Exception("Hull intersects full shoreline "+p.name);foreach(var mf in boat.GetComponentsInChildren<MeshFilter>().Where(m=>m.gameObject.activeInHierarchy&&m.sharedMesh!=null)){var vv=mf.sharedMesh.vertices;var tr=mf.sharedMesh.triangles;for(int i=0;i<tr.Length;i+=3)if(Land(mf.transform.TransformPoint((vv[tr[i]]+vv[tr[i+1]]+vv[tr[i+2]])/3)))throw new Exception("Hull face intersects shoreline");}foreach(var mf in p.Find("Plot_"+(Array.IndexOf(plots,p)+1)+"_Bridges092/Private_Port_Branch092").GetComponentsInChildren<MeshFilter>().Where(m=>m.gameObject.name=="Deck_Plank"))foreach(var v in mf.sharedMesh.vertices){var w=mf.transform.TransformPoint(v);if(p.InverseTransformPoint(w).x<-2&&Land(w))throw new Exception("Branch on full shoreline");}}
 var skins=root.Find("04_Boat_Skins_Showcase");var sp=new[]{new Vector2(-6,-10),new Vector2(6,-10),new Vector2(6,10),new Vector2(-6,10)}.Select(v=>skins.TransformPoint(new Vector3(v.x,0,v.y))).Select(v=>new Vector2(v.x,v.z)).ToList();float skinGap=plots.Min(p=>Clearance(shores[p],sp));if(skinGap<25)throw new Exception("Skin pier gap reduced");if(root.GetComponentsInChildren<Transform>(true).Any(t=>t.name=="Pebble_Cluster092")||root.GetComponentsInChildren<Collider>(true).Length>0||root.GetComponentsInChildren<Rigidbody>(true).Length>0)throw new Exception("Visual invariant failure");report.AppendLine("Verified all eight hull vertices + face centers against full foam shoreline, entire hull bounding footprint >=2.1m off main foam, branch corners on water. Min personal full shore gap="+minIslands.ToString("F2")+"m; personal shore to skin pier="+skinGap.ToString("F2")+"m. No runtime/physics changes.");Shot(new Vector3(150,230,-210),main.position,113,"15-angular-stones-overview093");Shot(plots[0].TransformPoint(new Vector3(-40,52,43)),plots[0].TransformPoint(new Vector3(-5,0,25)),30,"16-plot01-shore-clearance093");Shot(new Vector3(90,100,-25),main.TransformPoint(new Vector3(0,0,12)),60,"17-angular-main-island093");EditorSceneManager.MarkSceneDirty(s);if(!EditorSceneManager.SaveScene(s))throw new Exception("Save failed");File.WriteAllText("Docs/assets/lobby-island/ROCKS_SHORE093_VALIDATION.txt",report.ToString()+"Scene clean="+!s.isDirty);return report.ToString();}
