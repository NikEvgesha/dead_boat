using System;using System.IO;using System.Linq;using System.Text;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;
public static class RefineLobbyDocks089 {
 const string D="Assets/Prefabs/Workshop/LobbyIsland/";
 static Material oak,dark,brass,gravel;
 static Transform G(string n,Transform p,Vector3 v){var t=new GameObject(n).transform;t.SetParent(p,false);t.localPosition=v;return t;}
 static GameObject Box(string n,Transform p,Vector3 v,Vector3 size,Material m){var g=GameObject.CreatePrimitive(PrimitiveType.Cube);g.name=n;g.transform.SetParent(p,false);g.transform.localPosition=v;g.transform.localScale=size;g.GetComponent<Renderer>().sharedMaterial=m;UnityEngine.Object.DestroyImmediate(g.GetComponent<Collider>());return g;}
 static Bounds B(Transform t){var rs=t.GetComponentsInChildren<MeshRenderer>().Where(r=>r.enabled&&r.gameObject.activeInHierarchy).ToArray();var b=rs[0].bounds;foreach(var r in rs.Skip(1))b.Encapsulate(r.bounds);return b;}
 static void Place(Transform t,Transform p,Vector3 target){var b=B(t);var w=p.TransformPoint(target);t.position+=new Vector3(w.x-b.center.x,w.y-b.min.y,w.z-b.center.z);}
 static Transform Deck(string n,Transform p,Vector3 a,Vector3 b,float width){var delta=b-a;var t=G(n,p,(a+b)/2);t.localRotation=Quaternion.LookRotation(delta);int count=Mathf.CeilToInt(delta.magnitude/.5f);for(int i=0;i<count;i++)Box("Deck_Plank",t,new Vector3(0,-.12f,-delta.magnitude/2+(i+.5f)*delta.magnitude/count),new Vector3(width,.24f,delta.magnitude/count-.03f),oak);for(float z=-delta.magnitude/2;z<delta.magnitude/2;z+=3)foreach(float x in new[]{-width/2+.16f,width/2-.16f}){Box("Dock_Piling",t,new Vector3(x,-1.1f,z),new Vector3(.18f,3.3f,.18f),dark);Box("Brass_Post_Cap",t,new Vector3(x,.58f,z),new Vector3(.23f,.07f,.23f),brass);}return t;}
 static void Path(Transform p,Vector3 a,Vector3 b,float width){var t=Box("Compact_Arrival_Path089",p,(a+b)/2+Vector3.up*.065f,new Vector3(width,.13f,(b-a).magnitude),gravel).transform;t.localRotation=Quaternion.LookRotation(b-a);}
 static bool OnGrass(Transform island,Vector3 world){var mf=island.Find("Grass_Terrace").GetComponent<MeshFilter>();var p=mf.transform.InverseTransformPoint(world);var vv=mf.sharedMesh.vertices;var tr=mf.sharedMesh.triangles;var pt=new Vector2(p.x,p.z);for(int i=0;i<tr.Length;i+=3){var a=new Vector2(vv[tr[i]].x,vv[tr[i]].z);var b=new Vector2(vv[tr[i+1]].x,vv[tr[i+1]].z);var c=new Vector2(vv[tr[i+2]].x,vv[tr[i+2]].z);float den=(b.y-c.y)*(a.x-c.x)+(c.x-b.x)*(a.y-c.y);float u=((b.y-c.y)*(pt.x-c.x)+(c.x-b.x)*(pt.y-c.y))/den;float v=((c.y-a.y)*(pt.x-c.x)+(a.x-c.x)*(pt.y-c.y))/den;if(u>=-.001f&&v>=-.001f&&u+v<=1.001f)return true;}return false;}
 static void Shot(Vector3 pos,Vector3 target,float size,string file){var g=new GameObject("Temporary_Lobby089_Camera");var rt=new RenderTexture(1800,1200,24);var tx=new Texture2D(1800,1200,TextureFormat.RGB24,false);var old=RenderTexture.active;try{var c=g.AddComponent<Camera>();c.transform.position=pos;c.transform.LookAt(target);c.orthographic=true;c.orthographicSize=size;c.farClipPlane=500;c.clearFlags=CameraClearFlags.SolidColor;c.backgroundColor=new Color(.19f,.39f,.43f);c.targetTexture=rt;c.Render();RenderTexture.active=rt;tx.ReadPixels(new Rect(0,0,1800,1200),0,0);tx.Apply();File.WriteAllBytes("Docs/assets/lobby-island/"+file+".png",tx.EncodeToPNG());}finally{RenderTexture.active=old;UnityEngine.Object.DestroyImmediate(g);UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(tx);}}
 public static string ClearShoreApproach(){var root=GameObject.Find("Lobby_Island_Visual_Prototype").transform;var main=root.Find("01_Central_Island");var skins=root.Find("04_Boat_Skins_Showcase");oak=AssetDatabase.LoadAssetAtPath<Material>(D+"Oiled_Oak.mat");dark=AssetDatabase.LoadAssetAtPath<Material>(D+"Deep_Charcoal.mat");brass=AssetDatabase.LoadAssetAtPath<Material>(D+"Brass_Gold.mat");gravel=AssetDatabase.LoadAssetAtPath<Material>(D+"Pale_Gravel.mat");UnityEngine.Object.DestroyImmediate(root.Find("Skin_Pier_Shore_Connection089").gameObject);var old=main.Cast<Transform>().First(t=>t.name=="Compact_Arrival_Path089"&&t.localPosition.x>30);UnityEngine.Object.DestroyImmediate(old.gameObject);var a=main.TransformPoint(new Vector3(38,0,-8));var b=skins.TransformPoint(new Vector3(-5,0,-6));if(!OnGrass(main,a))throw new Exception("Approach off shore");Deck("Skin_Pier_Shore_Connection089",root,a,b,3);Path(main,new Vector3(38,0,-8),new Vector3(28,0,-2),3);Shot(new Vector3(95,165,-170),new Vector3(7,0,-3),92,"05-compact-overview089");Shot(new Vector3(35,42,-10),new Vector3(61,1,33),24,"06-skins-pier089");var s=UnityEngine.SceneManagement.SceneManager.GetActiveScene();EditorSceneManager.MarkSceneDirty(s);EditorSceneManager.SaveScene(s);File.AppendAllText("Docs/assets/lobby-island/DOCKS089_VALIDATION.txt","Final shore approach routed south of existing palm and shore rocks; shore start (38,0,27), deck end (60,0,29), length 22.09m.\n");return "Pier approach moved clear of palm/rocks and saved.";}
 public static string Run(){var s=UnityEngine.SceneManagement.SceneManager.GetActiveScene();if(s.path!="Assets/Scenes/LobbyIslandWorkshop.unity"||s.isDirty||EditorApplication.isPlayingOrWillChangePlaymode)throw new Exception("Saved prototype Edit Mode required");var root=GameObject.Find("Lobby_Island_Visual_Prototype").transform;if(root.Find("07_Compact_Connections089")!=null)throw new Exception("Already applied");oak=AssetDatabase.LoadAssetAtPath<Material>(D+"Oiled_Oak.mat");dark=AssetDatabase.LoadAssetAtPath<Material>(D+"Deep_Charcoal.mat");brass=AssetDatabase.LoadAssetAtPath<Material>(D+"Brass_Gold.mat");gravel=AssetDatabase.LoadAssetAtPath<Material>(D+"Pale_Gravel.mat");var report=new StringBuilder();
 var boats=root.GetComponentsInChildren<Transform>(true).Where(t=>t.name.StartsWith("Actual_B")).ToArray();if(boats.Length!=5)throw new Exception("Expected five visual boats");
 foreach(var boat in boats){foreach(string n in new[]{"TrailerDefault","Cube (1)","LoadedItemsSpawnPoint"}){var t=boat.Find(n);if(t!=null)t.gameObject.SetActive(false);}if(boat.Find("BoardDefault")==null||!boat.Find("BoardDefault").gameObject.activeSelf)throw new Exception("Front hull missing");report.AppendLine(boat.name+": front-only size="+B(boat).size.ToString("F3")+" scale="+boat.localScale);}
 var main=root.Find("01_Central_Island");var plots=root.Find("02_Personal_Islands_4_Visual_Samples").Cast<Transform>().ToArray();
 var oldBridges=root.Cast<Transform>().Where(t=>t.name=="Timber_Connected").ToArray();if(oldBridges.Length!=4)throw new Exception("Expected four inter-island bridges");float[] oldLength=oldBridges.Select(t=>t.GetComponentsInChildren<Transform>().Where(c=>c.name=="Deck_Plank").Sum(c=>c.localScale.z+.03f)).ToArray();
 foreach(var plot in plots){var v=plot.localPosition;v.z+=10;plot.localPosition=v;var dock=plot.Find("04_Private_Dock");foreach(var old in dock.Cast<Transform>().Where(t=>t.name=="Timber_Connected").ToArray())UnityEngine.Object.DestroyImmediate(old.gameObject);Deck("Compact_Private_Dock089",dock,new Vector3(0,0,-18),new Vector3(0,0,-42),2.6f);Deck("Compact_Private_Dock_Return089",dock,new Vector3(0,0,-42),new Vector3(11,0,-42),2.6f);var boat=dock.Find("Actual_BoardController_RenderOnly_1to1");Place(boat,dock,new Vector3(7,-1,-32));var b=B(boat);if(b.size.z>20||b.max.z>plot.position.z-21)throw new Exception("Front hull berth/shore clearance failed");dock.Find("Future_Boat_Spawn_Anchor").localPosition=boat.localPosition;var reserve=dock.Find("Berth_Reserved_10x32m");reserve.name="Berth_Reserved_FrontHull_10x20m";reserve.localPosition=new Vector3(7,0,-32);dock.Find("Future_Dock_Interaction_Anchor").localPosition=new Vector3(0,0,-29);}
 foreach(var old in oldBridges)UnityEngine.Object.DestroyImmediate(old.gameObject);foreach(var old in main.Cast<Transform>().Where(t=>t.name=="Arrival_Gravel_Path").ToArray())UnityEngine.Object.DestroyImmediate(old.gameObject);
 var connections=G("07_Compact_Connections089",root,Vector3.zero);
 for(int i=0;i<plots.Length;i++){var plot=plots[i];float side=Mathf.Sign(plot.localPosition.x);bool outer=i==0||i==3;var a=plot.TransformPoint(new Vector3(outer?-side*4:0,0,16));var b=main.TransformPoint(new Vector3(side*(outer?34:20),0,outer?-15:-25));for(int j=0;j<6&&!OnGrass(plot,a);j++)a=Vector3.Lerp(a,plot.position,.08f);for(int j=0;j<6&&!OnGrass(main,b);j++)b=Vector3.Lerp(b,main.position,.04f);if(!OnGrass(plot,a)||!OnGrass(main,b))throw new Exception("Bridge endpoint outside grass "+i+" "+a+" "+b);float length=(b-a).magnitude;if(length>=oldLength[i]*.9f)throw new Exception("Bridge not shortened");Deck("Short_Island_Bridge_"+(i+1),connections,root.InverseTransformPoint(a),root.InverseTransformPoint(b),2.4f);Path(plot,plot.InverseTransformPoint(a),new Vector3(1.5f,0,15),2.4f);Path(main,main.InverseTransformPoint(b),new Vector3(side*10,0,-13),2.4f);main.Find("Central_Arrival_Anchor_"+(i+1)).position=b;report.AppendLine(plot.name+": bridge "+oldLength[i].ToString("F2")+" -> "+length.ToString("F2")+"m, width 3 -> 2.4m; both endpoints inside actual grass triangles; whole plot moved +10m in z.");}
 // Move the entire skin-change station off the central island, to a separate pier at its right shore.
 var skins=main.Find("04_Boat_Skins_Showcase");skins.SetParent(root,true);skins.position=new Vector3(65,0,35);var deck=skins.Find("Boat_Exhibition_Deck");deck.localScale=new Vector3(12,.45f,20);var exhibit=skins.Find("Actual_Boat_RenderOnly_1to1");Place(exhibit,skins,new Vector3(1.2f,.5f,0));var eb=B(exhibit);if(eb.size.z>20||eb.size.x>8)throw new Exception("Exhibit outside compact pier");skins.Find("Skins_Header").localPosition=new Vector3(-5.75f,2.8f,7);skins.Find("Skins_Header").localRotation=Quaternion.Euler(0,90,0);skins.Find("Skins_Header").localScale=new Vector3(8,.8f,.2f);skins.Find("Skins_Sign").localPosition=new Vector3(-5.9f,2.8f,7);skins.Find("Skins_Sign").localRotation=Quaternion.Euler(0,90,0);skins.Find("Skins_Sign").GetComponent<TextMesh>().characterSize=.13f;skins.Find("Future_Skin_Purchase_Anchor").localPosition=new Vector3(-4.5f,.5f,0);skins.Find("Viewing_Route_3m").localPosition=new Vector3(-4.5f,.5f,-4);
 var start=main.TransformPoint(new Vector3(40,0,0));var end=skins.TransformPoint(new Vector3(-5,0,0));if(!OnGrass(main,start))throw new Exception("Skin pier approach not on shore");Deck("Skin_Pier_Shore_Connection089",root,root.InverseTransformPoint(start),root.InverseTransformPoint(end),3);Path(main,new Vector3(40,0,0),new Vector3(28,0,-2),3);foreach(float z in new[]{-9f,9f})foreach(float x in new[]{-5.5f,5.5f}){Box("Skin_Pier_Piling",skins,new Vector3(x,-1.5f,z),new Vector3(.28f,4,.28f),dark);Box("Skin_Pier_Bollard",skins,new Vector3(x,.55f,z),new Vector3(.4f,.65f,.4f),oak);}
 report.AppendLine("Skin-change visual station moved to separate right pier world (65,0,35), deck 14x32 -> 12x20m. 20m shore connection, front hull 1:1, clear left viewing route. Private longitudinal docks 34 ->24m for front-only boats.");
 if(root.GetComponentsInChildren<Collider>(true).Length!=0||root.GetComponentsInChildren<Rigidbody>(true).Length!=0||root.GetComponentsInChildren<Transform>(true).Any(t=>GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(t.gameObject)>0))throw new Exception("Visual prototype component violation");
 Shot(new Vector3(95,165,-170),new Vector3(7,0,-3),92,"05-compact-overview089");Shot(new Vector3(110,48,0),new Vector3(59,1,34),24,"06-skins-pier089");
 EditorSceneManager.MarkSceneDirty(s);AssetDatabase.SaveAssets();if(!EditorSceneManager.SaveScene(s))throw new Exception("Save failed");File.WriteAllText("Docs/assets/lobby-island/DOCKS089_VALIDATION.txt",report.ToString());return report.ToString();
 }
